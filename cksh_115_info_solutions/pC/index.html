pC. 完美生態照

TIOJ 原題

題解

先依照 x 由大到小排序，再者 O(n) 掃過去一次，維護雙指針 l, r，確保：

  1. 區間中各種生物至少一隻。（使用雜湊表計數）
  2. 區間不可以只取某位置上的某個生物。（生物可以重疊！）

又區間雙指針必定向右單調，故可以使用兩個單調佇列維護最大最小值，O(1) 計算該區間是否合法，合法則更新答案，接著 r 向右……，直到掃完全部。

程式碼

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using pll = pair<ll,ll>;

void solve();

int main(){
    int T = 1;
    cin.tie(0);
    ios::sync_with_stdio(0);
    // cin >> T;
    while(T--){
        solve();
    }
    return 0;
}

void solve(){
    ll i, j, i1, i2, t1, t2;
    struct ani{ll x, t, s;};
    ll n, k, d, l, r, ans = LLONG_MAX;
    cin >> n >> k >> d;
    vector<ani> v(n);
    for(i=0;i<n;i++) cin >> v[i].x >> v[i].t >> v[i].s;
    sort(v.begin(), v.end(), [](ani &a, ani &b){return a.x < b.x;});
    unordered_map<ll, ll> mp;
    deque<ll> ma, mi, dq;
    l = 0;
    for(r=0;r<n;r++){
        mp[v[r].t]++;
        while(!ma.empty()){
            if(v[r].s > v[ma.back()].s){
                ma.pop_back();
            }else{
                break;
            }
        }
        ma.push_back(r);
        while(!mi.empty()){
            if(v[r].s < v[mi.back()].s){
                mi.pop_back();
            }else{
                break;
            }
        }
        mi.push_back(r);
        if(mp.size()==k){
            while(r+1 < n){
                if(v[r+1].x == v[r].x){
                    r++;
                    mp[v[r].t]++;
                    while(!ma.empty()){
                        if(v[r].s > v[ma.back()].s){
                            ma.pop_back();
                        }else{
                            break;
                        }
                    }
                    ma.push_back(r);
                    while(!mi.empty()){
                        if(v[r].s < v[mi.back()].s){
                            mi.pop_back();
                        }else{
                            break;
                        }
                    }
                    mi.push_back(r);
                }else{
                    break;
                }
            }
            break;
        }
    }
    while(r<n){
        while(!ma.empty() && ma.front() < l) ma.pop_front();
        while(!mi.empty() && mi.front() < l) mi.pop_front();
        if(v[ma.front()].s - v[mi.front()].s <= d){
            ans = min(ans, v[r].x - v[l].x);
        }
        mp[v[l].t]--;
        if(!mp[v[l].t]) dq.push_back(v[l].t);
        l++;
        while(l < n){
            if(v[l].x == v[l-1].x){
                mp[v[l].t]--;
                if(!mp[v[l].t]) dq.push_back(v[l].t);
                l++;
            }
            else break;
        }
        if(l == n) break;
        if(!dq.empty()){
            for(r=r+1;r<n;r++){
                mp[v[r].t]++;
                while(!ma.empty()){
                    if(v[r].s > v[ma.back()].s){
                        ma.pop_back();
                    }else{
                        break;
                    }
                }
                ma.push_back(r);
                while(!mi.empty()){
                    if(v[r].s < v[mi.back()].s){
                        mi.pop_back();
                    }else{
                        break;
                    }
                }
                mi.push_back(r);
                while(!dq.empty()) if(mp[dq.front()]) dq.pop_front(); else break;
                if(dq.empty()){
                    while(r+1 < n){
                        if(v[r+1].x == v[r].x){
                            r++;
                            mp[v[r].t]++;
                            while(!ma.empty()){
                                if(v[r].s > v[ma.back()].s){
                                    ma.pop_back();
                                }else{
                                    break;
                                }
                            }
                            ma.push_back(r);
                            while(!mi.empty()){
                                if(v[r].s < v[mi.back()].s){
                                    mi.pop_back();
                                }else{
                                    break;
                                }
                            }
                            mi.push_back(r);
                        }else{
                            break;
                        }
                    }
                    break;
                }
            }
        }
    }

    for(j=1;j<=k;j++){
        if(!mp[j]){
            cout << (ans == LLONG_MAX ? -1 : ans);
            return;
        }
    }

    if(v[ma.front()].s - v[mi.front()].s <= d){
        ans = min(ans, v[r-1].x - v[l].x);
    }
    cout << (ans == LLONG_MAX ? -1 : ans);
    return;
}
