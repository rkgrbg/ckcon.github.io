模擬，記得紀錄 vis 的不可以只有位置，還要有方向，因為方向不同傳出去的就不同。

程式碼

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using pll = pair<ll,ll>;

void solve();

int main(){
    int T = 1;
    cin.tie(0);
    ios::sync_with_stdio(0);
    // cin >> T;
    while(T--){
        solve();
    }
    return 0;
}

void solve(){
    ll i, j, k, l = 0, i1, i2, t1, t2;
    ll n, m;
    cin >> n >> m;
    deque<pair<pll, char>> dq;
    char c;
    cin >> t1 >> t2 >> c;
    dq.push_back({{t1-1,t2-1},c});
    vector<vector<int>> g(n, vector<int>(m));
    for(i=0;i<n;i++) for(j=0;j<m;j++) cin >> g[i][j];
    
    vector<vector<vector<bool>>> vis(n, vector<vector<bool>>(m, vector<bool>(5, 0)));
    
    while(!dq.empty()){
        i = dq.front().first.first;
        j = dq.front().first.second;
        c = dq.front().second;
        dq.pop_front();
        
        if(c == 'U' && vis[i][j][0]) continue;
        if(c == 'D' && vis[i][j][1]) continue;
        if(c == 'L' && vis[i][j][2]) continue;
        if(c == 'R' && vis[i][j][3]) continue;
        if(!vis[i][j][4]) l++;
        if(c == 'U') vis[i][j][0] = 1, vis[i][j][4] = 1;
        if(c == 'D') vis[i][j][1] = 1, vis[i][j][4] = 1;
        if(c == 'L') vis[i][j][2] = 1, vis[i][j][4] = 1;
        if(c == 'R') vis[i][j][3] = 1, vis[i][j][4] = 1;

        switch(g[i][j]){
            case 0 :{
                if(c == 'U' && i-1>=0) dq.push_back({{i-1,j},'U'});
                if(c == 'D' && i+1< n) dq.push_back({{i+1,j},'D'});
                if(c == 'L' && j-1>=0) dq.push_back({{i,j-1},'L'});
                if(c == 'R' && j+1< m) dq.push_back({{i,j+1},'R'});
                break;
            }
            case 1 :{
                break;
            }
            case 2 :{
                if(c == 'L' && i-1>=0) dq.push_back({{i-1,j},'U'});
                if(c == 'R' && i+1< n) dq.push_back({{i+1,j},'D'});
                if(c == 'U' && j-1>=0) dq.push_back({{i,j-1},'L'});
                if(c == 'D' && j+1< m) dq.push_back({{i,j+1},'R'});
                break;
            }
            case 3 :{
                if(c == 'R' && i-1>=0) dq.push_back({{i-1,j},'U'});
                if(c == 'L' && i+1< n) dq.push_back({{i+1,j},'D'});
                if(c == 'D' && j-1>=0) dq.push_back({{i,j-1},'L'});
                if(c == 'U' && j+1< m) dq.push_back({{i,j+1},'R'});
                break;
            }
            case 4 :{
                if(c == 'L' && i-1>=0) dq.push_back({{i-1,j},'U'});
                if(c == 'R' && i+1< n) dq.push_back({{i+1,j},'D'});
                if(c == 'U' && j-1>=0) dq.push_back({{i,j-1},'L'});
                if(c == 'D' && j+1< m) dq.push_back({{i,j+1},'R'});
                if(c == 'U' && i-1>=0) dq.push_back({{i-1,j},'U'});
                if(c == 'D' && i+1< n) dq.push_back({{i+1,j},'D'});
                if(c == 'L' && j-1>=0) dq.push_back({{i,j-1},'L'});
                if(c == 'R' && j+1< m) dq.push_back({{i,j+1},'R'});
                break;
            }
            case 5 :{
                if(c == 'R' && i-1>=0) dq.push_back({{i-1,j},'U'});
                if(c == 'L' && i+1< n) dq.push_back({{i+1,j},'D'});
                if(c == 'D' && j-1>=0) dq.push_back({{i,j-1},'L'});
                if(c == 'U' && j+1< m) dq.push_back({{i,j+1},'R'});
                if(c == 'U' && i-1>=0) dq.push_back({{i-1,j},'U'});
                if(c == 'D' && i+1< n) dq.push_back({{i+1,j},'D'});
                if(c == 'L' && j-1>=0) dq.push_back({{i,j-1},'L'});
                if(c == 'R' && j+1< m) dq.push_back({{i,j+1},'R'});
                break;
            }
        }
    }

    cout << l;
    return;
}
