pB. 天作之合

TIOJ 原題
載入題目內容中

題解

符合以下任一條件者為可分：

  1. 存在兩個連續的 0，等價存在三個連續相同的前綴和。
  2. 存在一個和為 0 的區間，且其大小≧2且＜n（不可全選），等價存在兩個相同但不相鄰的前綴和，或存在一個值為 0 的非全選前綴和。

又需處理環狀，故可以複製一份在其後面，並紀錄有效生命週期：

  1. 存在兩個連續的 0，等價對於任意項，存在三個連續相同的前綴和。
  2. 存在一個和為 0 的區間，且其大小≧2且＜n（不可全選），等價對於索引前方至多 n 項，存在一個與其相同但不相鄰的前綴和；或對於前 n−1 項，存在一個值為 0 的前綴和。

程式碼

#include <bits/stdc++.h>
using namespace std;
using ll = long long;
using pll = pair<ll,ll>;

void solve();

int main(){
    int T = 1;
    cin.tie(0);
    ios::sync_with_stdio(0);
    // cin >> T;
    while(T--){
        solve();
    }
    return 0;
}

void solve(){
    ll i, j, k, l = 0, i1, i2, t1, t2;
    ll n;
    cin >> n;
    vector<ll> v(n), ps(2*n+1, 0);
    for(i=0;i<n;i++) cin >> v[i];
    for(i=1;i<=2*n;i++){
        ps[i] = ps[i-1] + v[(i-1)%n];
    }
    
    unordered_map<ll, deque<ll>> mp;
    for(i=1;i<2*n+1;i++){
        if(i<n && !ps[i]){
            cout << "YES";
            return;
        }
        if(!mp[ps[i]].empty() && ps[i-1] != ps[i]){
            while(!mp[ps[i]].empty() && mp[ps[i]].front()<= i-n) mp[ps[i]].pop_front();
            if(!mp[ps[i]].empty()){
                cout << "YES";
                return;
            }
        }
        if(i<2*n && ps[i] == ps[i-1] && ps[i] == ps[i+1]){
            cout << "YES";
            return;
        }
        mp[ps[i]].push_back(i);
    }

    cout << "NO";
    return;
}
